          package com.projectmediay.app;

          import android.app.Notification;
          import android.app.NotificationChannel;
          import android.app.NotificationManager;
          import android.app.PendingIntent;
          import android.app.Service;
          import android.content.Context;
          import android.content.Intent;
          import android.content.pm.ServiceInfo;
          import android.os.Build;
          import android.os.Handler;
          import android.os.IBinder;
          import android.os.Looper;
          import android.os.PowerManager;

          public final class MediaYExportService extends Service {
              public static final String ACTION_START = "com.projectmediay.app.EXPORT_START";
              private static final String CHANNEL_ID = "mediay-export";
              private static final int NOTIFICATION_ID = 26024;
              // Safety net: if the web app never reports that the job ended, stop by itself instead of
              // holding the CPU awake and showing a stale notification forever.
              private static final long MAX_RUN_MS = 3L * 60L * 60L * 1000L;

              private PowerManager.WakeLock wakeLock;
              private final Handler handler = new Handler(Looper.getMainLooper());
              private final Runnable stopWhenExpired = new Runnable() {
                  @Override
                  public void run() {
                      stopSelf();
                  }
              };

              @Override
              public void onCreate() {
                  super.onCreate();
                  if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
                      NotificationChannel channel = new NotificationChannel(
                          CHANNEL_ID, "MediaY export", NotificationManager.IMPORTANCE_LOW);
                      channel.setDescription("Shows when MediaY is rendering a video.");
                      NotificationManager manager = getSystemService(NotificationManager.class);
                      if (manager != null) manager.createNotificationChannel(channel);
                  }
              }

              @Override
              public int onStartCommand(Intent intent, int flags, int startId) {
                  // A foreground service must call startForeground promptly after startForegroundService,
                  // so the notification is always built first.
                  Notification.Builder builder = Build.VERSION.SDK_INT >= Build.VERSION_CODES.O
                      ? new Notification.Builder(this, CHANNEL_ID)
                      : new Notification.Builder(this);
                  builder.setSmallIcon(getApplicationInfo().icon)
                      .setContentTitle("MediaY is working")
                      .setContentText("Rendering your video. Tap to return to MediaY.")
                      .setCategory(Notification.CATEGORY_PROGRESS)
                      .setPriority(Notification.PRIORITY_LOW)
                      .setOngoing(true)
                      .setOnlyAlertOnce(true)
                      .setProgress(0, 0, true);

                  Intent launchIntent = getPackageManager().getLaunchIntentForPackage(getPackageName());
                  if (launchIntent != null) {
                      int pendingFlags = PendingIntent.FLAG_UPDATE_CURRENT;
                      if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
                          pendingFlags |= PendingIntent.FLAG_IMMUTABLE;
                      }
                      builder.setContentIntent(PendingIntent.getActivity(this, 0, launchIntent, pendingFlags));
                  }

                  Notification notification = builder.build();
                  if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
                      startForeground(NOTIFICATION_ID, notification, ServiceInfo.FOREGROUND_SERVICE_TYPE_DATA_SYNC);
                  } else {
                      startForeground(NOTIFICATION_ID, notification);
                  }

                  // Without a wake lock Android may suspend the CPU as soon as the screen turns off,
                  // even while a foreground service is running.
                  if (wakeLock == null) {
                      PowerManager power = (PowerManager) getSystemService(Context.POWER_SERVICE);
                      if (power != null) {
                          wakeLock = power.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "MediaY:export");
                          wakeLock.setReferenceCounted(false);
                          wakeLock.acquire(MAX_RUN_MS);
                      }
                  }
                  handler.removeCallbacks(stopWhenExpired);
                  handler.postDelayed(stopWhenExpired, MAX_RUN_MS);

                  // NOT_STICKY: if Android kills the process, do not resurrect an "export in progress"
                  // notification for an export that no longer exists.
                  return START_NOT_STICKY;
              }

              @Override
              public void onDestroy() {
                  handler.removeCallbacks(stopWhenExpired);
                  if (wakeLock != null && wakeLock.isHeld()) {
                      wakeLock.release();
                  }
                  wakeLock = null;
                  stopForeground(true);
                  super.onDestroy();
              }

              @Override
              public IBinder onBind(Intent intent) {
                  return null;
              }
          }

          package com.projectmediay.app;

          import android.Manifest;
          import android.content.ContentResolver;
          import android.content.ContentValues;
          import android.media.MediaScannerConnection;
          import android.net.Uri;
          import android.os.Build;
          import android.os.Environment;
          import android.provider.MediaStore;

          import com.getcapacitor.JSObject;
          import com.getcapacitor.PermissionState;
          import com.getcapacitor.Plugin;
          import com.getcapacitor.PluginCall;
          import com.getcapacitor.PluginMethod;
          import com.getcapacitor.annotation.CapacitorPlugin;
          import com.getcapacitor.annotation.Permission;
          import com.getcapacitor.annotation.PermissionCallback;

          import java.io.BufferedInputStream;
          import java.io.BufferedOutputStream;
          import java.io.File;
          import java.io.FileInputStream;
          import java.io.FileOutputStream;
          import java.io.IOException;
          import java.io.InputStream;
          import java.io.OutputStream;

          @CapacitorPlugin(
              name = "MediaYMediaStore",
              permissions = {
                  @Permission(alias = "legacyStorage", strings = { Manifest.permission.WRITE_EXTERNAL_STORAGE })
              }
          )
          public final class MediaYMediaStorePlugin extends Plugin {
              private Uri streamingUri;
              private OutputStream streamingOutput;
              private String streamingFileName;
              private long streamingBytes;

              @PluginMethod
              public synchronized void beginStreamingSave(PluginCall call) {
                  if (Build.VERSION.SDK_INT < Build.VERSION_CODES.Q) {
                      call.reject("Streaming MediaStore export requires Android 10 or newer.");
                      return;
                  }
                  String fileName = call.getString("fileName");
                  if (fileName == null || !fileName.matches("ProjectMedia-Y-[0-9]+\\.mp4")) {
                      call.reject("Invalid exported video filename.");
                      return;
                  }
                  abortStreamingSaveInternal();
                  ContentResolver resolver = getContext().getContentResolver();
                  try {
                      ContentValues values = new ContentValues();
                      values.put(MediaStore.Video.Media.DISPLAY_NAME, fileName);
                      values.put(MediaStore.Video.Media.MIME_TYPE, "video/mp4");
                      values.put(MediaStore.Video.Media.RELATIVE_PATH, Environment.DIRECTORY_MOVIES + "/MediaY");
                      values.put(MediaStore.Video.Media.IS_PENDING, 1);
                      Uri uri = resolver.insert(MediaStore.Video.Media.EXTERNAL_CONTENT_URI, values);
                      if (uri == null) throw new IOException("Android did not create a Movies/MediaY media entry.");
                      // Track the pending row before opening its stream so every failure path can delete it.
                      streamingUri = uri;
                      streamingFileName = fileName;
                      OutputStream output = resolver.openOutputStream(uri, "w");
                      if (output == null) {
                          throw new IOException("Android did not open the destination file.");
                      }
                      streamingOutput = new BufferedOutputStream(output, 256 * 1024);
                      streamingBytes = 0L;
                      JSObject result = new JSObject();
                      result.put("fileName", fileName);
                      result.put("location", "Movies/MediaY");
                      call.resolve(result);
                  } catch (Exception error) {
                      abortStreamingSaveInternal();
                      call.reject("Could not start streaming export: " + error.getMessage(), error);
                  }
              }

              @PluginMethod
              public synchronized void appendStreamingChunk(PluginCall call) {
                  String encoded = call.getString("data");
                  if (streamingOutput == null || encoded == null || encoded.isEmpty()) {
                      call.reject("No active streaming export or empty chunk.");
                      return;
                  }
                  try {
                      byte[] chunk = android.util.Base64.decode(encoded, android.util.Base64.DEFAULT);
                      if (chunk.length == 0 || chunk.length > 4 * 1024 * 1024) {
                          throw new IOException("Export chunk is outside the allowed size.");
                      }
                      streamingOutput.write(chunk);
                      streamingBytes += chunk.length;
                      JSObject result = new JSObject();
                      result.put("bytesWritten", chunk.length);
                      result.put("totalBytes", streamingBytes);
                      call.resolve(result);
                  } catch (Exception error) {
                      abortStreamingSaveInternal();
                      call.reject("Could not write export chunk: " + error.getMessage(), error);
                  }
              }

              @PluginMethod
              public synchronized void finishStreamingSave(PluginCall call) {
                  if (streamingOutput == null || streamingUri == null || streamingBytes <= 0L) {
                      abortStreamingSaveInternal();
                      call.reject("The streamed MP4 is empty or no export is active.");
                      return;
                  }
                  Uri uri = streamingUri;
                  String fileName = streamingFileName;
                  long bytes = streamingBytes;
                  try {
                      streamingOutput.flush();
                      streamingOutput.close();
                      streamingOutput = null;
                      ContentValues published = new ContentValues();
                      published.put(MediaStore.Video.Media.IS_PENDING, 0);
                      if (getContext().getContentResolver().update(uri, published, null, null) <= 0) {
                          throw new IOException("Android could not publish the exported video to Gallery.");
                      }
                      streamingUri = null;
                      streamingFileName = null;
                      streamingBytes = 0L;
                      JSObject result = new JSObject();
                      result.put("fileName", fileName);
                      result.put("location", "Movies/MediaY");
                      result.put("uri", uri.toString());
                      result.put("bytes", bytes);
                      call.resolve(result);
                  } catch (Exception error) {
                      abortStreamingSaveInternal();
                      call.reject("Could not finish streaming export: " + error.getMessage(), error);
                  }
              }

              @PluginMethod
              public synchronized void abortStreamingSave(PluginCall call) {
                  abortStreamingSaveInternal();
                  call.resolve();
              }

              private void abortStreamingSaveInternal() {
                  if (streamingOutput != null) {
                      try { streamingOutput.close(); } catch (Exception ignored) {}
                  }
                  streamingOutput = null;
                  if (streamingUri != null) {
                      try { getContext().getContentResolver().delete(streamingUri, null, null); } catch (Exception ignored) {}
                  }
                  streamingUri = null;
                  streamingFileName = null;
                  streamingBytes = 0L;
              }

              @PluginMethod
              public void saveVideo(PluginCall call) {
                  if (Build.VERSION.SDK_INT < Build.VERSION_CODES.Q
                      && getPermissionState("legacyStorage") != PermissionState.GRANTED) {
                      requestPermissionForAlias("legacyStorage", call, "legacyStoragePermissionCallback");
                      return;
                  }
                  saveVideoInternal(call);
              }

              @PermissionCallback
              private void legacyStoragePermissionCallback(PluginCall call) {
                  if (Build.VERSION.SDK_INT < Build.VERSION_CODES.Q
                      && getPermissionState("legacyStorage") != PermissionState.GRANTED) {
                      call.reject("Storage permission was denied. MediaY could not save the MP4.");
                      return;
                  }
                  saveVideoInternal(call);
              }

              private void saveVideoInternal(PluginCall call) {
                  String fileName = call.getString("fileName");
                  if (fileName == null || !fileName.matches("ProjectMedia-Y-[0-9]+\\.mp4")) {
                      call.reject("Invalid exported video filename.");
                      return;
                  }

                  File source = new File(getContext().getCacheDir(), fileName);
                  if (!source.isFile() || source.length() <= 0L) {
                      call.reject("The temporary exported MP4 is missing or empty.");
                      return;
                  }

                  ContentResolver resolver = getContext().getContentResolver();
                  Uri insertedUri = null;
                  File legacyFile = null;
                  try {
                      if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
                          ContentValues values = new ContentValues();
                          values.put(MediaStore.Video.Media.DISPLAY_NAME, fileName);
                          values.put(MediaStore.Video.Media.MIME_TYPE, "video/mp4");
                          values.put(
                              MediaStore.Video.Media.RELATIVE_PATH,
                              Environment.DIRECTORY_MOVIES + "/MediaY"
                          );
                          values.put(MediaStore.Video.Media.IS_PENDING, 1);
                          insertedUri = resolver.insert(MediaStore.Video.Media.EXTERNAL_CONTENT_URI, values);
                          if (insertedUri == null) {
                              throw new IOException("Android did not create a Movies/MediaY media entry.");
                          }

                          try (InputStream input = new BufferedInputStream(new FileInputStream(source));
                               OutputStream output = resolver.openOutputStream(insertedUri, "w")) {
                              if (output == null) throw new IOException("Android did not open the destination file.");
                              copy(input, output);
                          }

                          ContentValues published = new ContentValues();
                          published.put(MediaStore.Video.Media.IS_PENDING, 0);
                          if (resolver.update(insertedUri, published, null, null) <= 0) {
                              throw new IOException("Android could not publish the exported video to Gallery.");
                          }
                      } else {
                          File movies = Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_MOVIES);
                          File folder = new File(movies, "MediaY");
                          if (!folder.isDirectory() && !folder.mkdirs() && !folder.isDirectory()) {
                              throw new IOException("Could not create the Movies/MediaY folder.");
                          }
                          legacyFile = new File(folder, fileName);
                          if (legacyFile.exists()) throw new IOException("A video with this name already exists.");
                          try (InputStream input = new BufferedInputStream(new FileInputStream(source));
                               OutputStream output = new BufferedOutputStream(new FileOutputStream(legacyFile))) {
                              copy(input, output);
                          }
                          MediaScannerConnection.scanFile(
                              getContext(),
                              new String[] { legacyFile.getAbsolutePath() },
                              new String[] { "video/mp4" },
                              null
                          );
                      }

                      JSObject result = new JSObject();
                      result.put("fileName", fileName);
                      result.put("location", "Movies/MediaY");
                      result.put("uri", insertedUri != null ? insertedUri.toString() : legacyFile.getAbsolutePath());
                      result.put("bytes", source.length());
                      call.resolve(result);
                  } catch (Exception error) {
                      if (insertedUri != null) {
                          try { resolver.delete(insertedUri, null, null); } catch (Exception ignored) {}
                      }
                      if (legacyFile != null && legacyFile.exists()) {
                          try { legacyFile.delete(); } catch (Exception ignored) {}
                      }
                      call.reject("Could not save the MP4 to Movies/MediaY: " + error.getMessage(), error);
                  }
              }

              private static void copy(InputStream input, OutputStream output) throws IOException {
                  byte[] buffer = new byte[256 * 1024];
                  int count;
                  while ((count = input.read(buffer)) != -1) {
                      output.write(buffer, 0, count);
                  }
                  output.flush();
              }
          }

# Instructions for AI coding agents

## Read first

- `docs/timeline-trim-architecture.md` — timeline timing model, trim invariants, and the intended transactional edit architecture.
- `docs/performance-triage.md` — what has been optimized, known latency limits, release gates, and the prioritized next-agent task list.

## Project constraints

MediaY is a browser-first video editor with a Capacitor Android build. Project media is stored locally in IndexedDB; do not introduce an upload/server dependency for editing or captions without explicitly designing the privacy model and asking for user consent.

The main video lane is a serialized list. Audio and overlays use independent absolute timeline positions. Do not implement global ripple by shifting every `start` field: items crossing a ripple range need source-correct splitting, and video overlays must preserve `sourceIn`.

Treat timeline edits as transactions:
1. Capture stable target IDs and source/timeline times.
2. Validate all affected ranges before mutation.
3. Commit one complete valid state as one Undo operation, or leave the project unchanged.
4. Keep preview timing, audio timing, export timing, autosave, and Undo/Redo consistent.

The FFmpeg export runs in WebAssembly (unless the device-encoder path in `docs/fast-export.md` handles the video). Memory usage is a first-class constraint on Android. Keep source/overlay asset caches bounded and clean generated FFmpeg virtual-filesystem files on both success and failure.

Auto captions use Transformers.js Whisper. A first run downloads the model; browser model caching is enabled by the library. Caption inference is chunked, but source audio decoding still uses whole-buffer `decodeAudioData`, which is a known long-file latency bottleneck. Do not claim performance improvements without recording measurements on real media.

## Required validation

For JavaScript changes:
```sh
node --check app.js
node --check setup.js
node --check sw.js
```

For browser and media-worker regressions, install Playwright plus Chromium and run:
```sh
python test_smoke.py
python test_ffmpeg_worker.py
```

The GitHub Actions **MediaY smoke tests** workflow runs those checks for pull requests. **Build APK** compiles the Android debug APK and uploads it as `MediaY-debug-apk`. The build also stamps a monotonically increasing Android version code from the workflow run number. A worker smoke test is not a substitute for an app-level export test; add real end-to-end export coverage as the next agent task.

Do not merge if the same PR head has a failing/pending required check. After code changes, verify the Android APK build as well as browser tests.

## Releases

A push/merge to `main` builds and uploads only a **debug APK**. It does not publish a GitHub Release automatically.

Production release publication is explicit: use **Actions → Build APK → Run workflow** and opt in to **Build, sign, and publish a production GitHub Release**. Before doing so, the repository owner must configure these Actions secrets:

- `ANDROID_KEYSTORE_BASE64`: Base64-encoded contents of the private release keystore.
- `ANDROID_KEYSTORE_PASSWORD`: keystore password.
- `ANDROID_KEY_ALIAS`: signing key alias.
- `ANDROID_KEY_PASSWORD`: key password.

Never commit the keystore, print the secrets, or replace an established app signing key without a deliberate migration plan. The workflow decodes the keystore into the ephemeral runner, builds `assembleRelease`, verifies the resulting APK with Android `apksigner`, and publishes the signed APK plus a SHA-256 checksum. Release tags are `v1.0.<workflow-run-number>`; Android `versionCode` uses the same positive run number. If a signing secret is absent, the release job should fail rather than publish a debug or unsigned APK. Debug APKs and production-signed APKs are different signing identities and may not update each other in place.

## Known scope gaps

- Physical-device export performance has not yet been benchmarked.
- The device-encoder (WebCodecs) export path in `mediay-fast.js` is new and has not run on a physical Android device. See `docs/fast-export.md`; it falls back to the FFmpeg-only path on any problem. CI exercises H.264 WebCodecs output when its Chromium build supports it, but Android WebView hardware support and speed still require a phone test.
- No full app-level automated test validates export duration, frame/audio sync, and repeat export after Undo/Redo.
- Export cancellation is not implemented. Export/caption stage timing diagnostics are available in the Export pane, but physical-device benchmark data is still missing.
- Whole-media caption audio decoding is still slow for long videos.
- Multi-thread FFmpeg has not been enabled; it requires cross-origin isolation / `SharedArrayBuffer` verification and may increase memory use.
- Global ripple-across-tracks, slip, and roll edits are not yet implemented.
